
CREATE TABLE IF NOT EXISTS usuario (
    identificacion INT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(100) NOT NULL UNIQUE,
    activo VARCHAR(10) NOT NULL DEFAULT 'S'
);

CREATE TABLE IF NOT EXISTS libro (
    isbn VARCHAR(20) PRIMARY KEY,
    titulo VARCHAR(150) NOT NULL,
    autor VARCHAR(100) NOT NULL,
    anio INT NULL
);

CREATE TABLE IF NOT EXISTS ejemplar (
    codigo VARCHAR(30) PRIMARY KEY,
    libro_isbn VARCHAR(20) NOT NULL,
    estado VARCHAR(20) NOT NULL DEFAULT 'Disponible',
    CONSTRAINT fk_ejemplar_libro 
        FOREIGN KEY (libro_isbn) REFERENCES libro(isbn) 
        ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE IF NOT EXISTS prestamo (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    ejemplar_codigo VARCHAR(30) NOT NULL,
    fecha_prestamo DATE NOT NULL,
    fecha_limite DATE NOT NULL,
    fecha_devolucion DATE NULL,
    estado VARCHAR(20) NOT NULL DEFAULT 'Activo',
    CONSTRAINT fk_prestamo_usuario 
        FOREIGN KEY (usuario_id) REFERENCES usuario(identificacion) 
        ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT fk_prestamo_ejemplar 
        FOREIGN KEY (ejemplar_codigo) REFERENCES ejemplar(codigo) 
        ON DELETE RESTRICT ON UPDATE CASCADE
);
